public class DemoGrpcClient {
  private final ManagedChannel channel;
  protect DemoGrpcClient() {
        ThreadPoolExecutor threadPoolExecutor = new ThreadPoolExecutor(core, core, 10L, TimeUnit.SECONDS,
                new LinkedBlockingQueue<>(10000),
                new ThreadFactoryBuilder().setDaemon(true).setNameFormat("nacos-grpc-client-executor-%d")
                        .build());
        threadPoolExecutor.allowCoreThreadTimeOut(true);
        channel = ManagedChannelBuilder
                .forAddress(host, port)
                .compressorRegistry(CompressorRegistry.getDefaultInstance())
                .decompressorRegistry(DecompressorRegistry.getDefaultInstance())
                .keepAliveTime(6 * 1000 * 1000, TimeUnit.MILLISECONDS)
                .executor(threadPoolExecutor)
                .defaultLoadBalancingPolicy("round_robin")
                .usePlaintext()
                .maxInboundMessageSize(DEFAULT_MAX_INBOUND_MESSAGE_SIZE)
                .disableRetry()
                .disableServiceConfigLookUp()
                .build();
}
