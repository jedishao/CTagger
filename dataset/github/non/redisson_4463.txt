```java
 @Bean(destroyMethod = "shutdown")
    @ConditionalOnProperty(name = "redis.model", havingValue = "cluster", matchIfMissing = false)
    public RedissonClient redisson() {
        log.info("===> 加载集群模式的redisson ...");
        String[] nodes = redisProperties.getCluster().getNodes().stream()
                .map(node -> REDIS_PROTOCOL_PREFIX + node).toArray(String[]::new);

        Config config = new Config();
        ClusterServersConfig clusterServersConfig = config
                .setCodec(JsonJacksonCodec.INSTANCE)
                .setThreads(redissonProperties.getThreads())
                .setNettyThreads(redissonProperties.getNettyThreads())
                .useClusterServers();
        clusterServersConfig
                .setPassword(redisProperties.getPassword())//设置密码
                .setPingConnectionInterval(redissonProperties.getPingConnectionInterval()) // ping连接间隔
                .setScanInterval(redissonProperties.getScanInterval()) // 主节点变化扫描间隔时间
                .setRetryInterval(redissonProperties.getRetryInterval()) // 命令重试发送时间间隔
                .setTimeout(redissonProperties.getTimeout()) // 命令等待超时
                .setSubscriptionConnectionPoolSize(redissonProperties.getSubscriptionConnectionPoolSize()) // 从节点发布和订阅连接池大小
                .setSubscriptionConnectionMinimumIdleSize(redissonProperties.getSubscriptionConnectionMinimumIdleSize()) // 从节点发布和订阅连接的最小空闲连接数
                .setSlaveConnectionPoolSize(redissonProperties.getSlaveConnectionPoolSize()) // 从节点连接池大小
                .setSlaveConnectionMinimumIdleSize(redissonProperties.getSlaveConnectionMinimumIdleSize()) //从节点最小空闲连接数
                .setMasterConnectionPoolSize(redissonProperties.getMasterConnectionPoolSize()) // 主节点连接池大小
                .setMasterConnectionMinimumIdleSize(redissonProperties.getMasterConnectionMinimumIdleSize()); // 主节点最小空闲连接数
//        if (!SpringUtil.getBean(RedisKeyConcatComponent.class).isProdOrUat()) {
//            clusterServersConfig.setNameMapper(new CustomNameMapper());
//        }
        clusterServersConfig
                .addNodeAddress(nodes)
                .setConnectTimeout(redissonProperties.getConnectTimeout())
                .setPassword(redisProperties.getPassword());
        return Redisson.create(config);
    }

    /**
     * 集群模式下优先加载RedissonConfig ,加载lettuceConfig集群下redis key过期监听会失败
     */
    @Bean
    @Order(Ordered.LOWEST_PRECEDENCE - 3)
    public RedissonConnectionFactory redissonConnectionFactory(RedissonClient redisson) {
        return new RedissonConnectionFactory(redisson);
    }

    @Bean
    public RedisMessageListenerContainer redisMessageListenerContainer(RedisConnectionFactory redisConnectionFactory) {
        RedisMessageListenerContainer container = new RedisMessageListenerContainer();
        container.setConnectionFactory(redisConnectionFactory);
        return container;
    }
```

```java
@Data
@ConfigurationProperties(prefix = "redisson.cluster")
public class RedissonProperties {

    /**
     * ping连接间隔
     */
    private Integer pingConnectionInterval = 30000;
    /**
     * 从节点最小空闲连接数
     * 多从节点的环境里，每个 从服务节点里用于普通操作（非 发布和订阅）的最小保持连接数（长连接）。长期保持一定数量的连接有利于提高瞬时读取反映速度。
     */
    private Integer slaveConnectionMinimumIdleSize = 24;

    /**
     * 从节点连接池大小
     * 多从节点的环境里，每个 从服务节点里用于普通操作（非 发布和订阅）连接的连接池最大容量。连接池的连接数量自动弹性伸缩。
     */
    private Integer slaveConnectionPoolSize = 64;

    /**
     * 主节点最小空闲连接数
     * 多从节点的环境里，每个 主节点的最小保持连接数（长连接）。长期保持一定数量的连接有利于提高瞬时写入反应速度。
     */
    private Integer masterConnectionMinimumIdleSize = 24;

    /**
     * 主节点连接池大小
     * 主节点的连接池最大容量。连接池的连接数量自动弹性伸缩。
     */
    private Integer masterConnectionPoolSize = 64;

    /**
     * 从节点发布和订阅连接的最小空闲连接数
     * 多从节点的环境里，每个 从服务节点里用于发布和订阅连接的最小保持连接数（长连接）。Redisson内部经常通过发布和订阅来实现许多功能。长期保持一定数量的发布订阅连接是必须的。
     */
    private Integer subscriptionConnectionMinimumIdleSize = 1;

    /**
     * 从节点发布和订阅连接池大小
     * 多从节点的环境里，每个 从服务节点里用于发布和订阅连接的连接池最大容量。连接池的连接数量自动弹性伸缩。
     */
    private Integer subscriptionConnectionPoolSize = 50;

    /**
     * 主节点变化扫描间隔时间
     * 对主节点变化节点状态扫描的时间间隔。单位是毫秒。
     *
     */
    private Integer scanInterval = 5000;

    /**
     * Netty线程池数量
     * 这个线程池数量是在一个Redisson实例内，被其创建的所有分布式数据类型和服务，以及底层客户端所一同共享的线程池里保存的线程数量。
     */
    private Integer nettyThreads = 32;

    /**
     * 命令重试发送时间间隔，单位：毫秒
     * 在某个节点执行相同或不同命令时，连续 失败 failedAttempts（执行失败最大次数） 时，该节点将被从可用节点列表里清除，直到 reconnectionTimeout（重新连接时间间隔） 超时以后再次尝试。
     */
    private Integer retryInterval = 1500;
    private int retryAttempts = 3;

    /**
     * 命令等待超时，单位：毫秒
     * 等待节点回复命令的时间。该时间从命令发送成功时开始计时。
     */
    private Integer timeout = 3000;

    /**
     * 线程池数量
     * 这个线程池数量被所有RTopic对象监听器，RRemoteService调用者和RExecutorService任务共同共享。
     */
    private Integer threads = 16;

    private int failedSlaveReconnectionInterval = 3000;

    private int failedSlaveCheckInterval = 180000;

    private ReadMode readMode = ReadMode.SLAVE;
    private SubscriptionMode subscriptionMode = SubscriptionMode.MASTER;
    private long dnsMonitoringInterval = 5000;
    private int connectTimeout = 10000;
    private int idleConnectionTimeout = 10000;
}
```
