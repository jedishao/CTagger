### Steps to reproduce the bug
Grpc service definition:
```protobuf
syntax = "proto3";

option java_multiple_files = true;
option java_package = "test";
option java_outer_classname = "TestContract";

package test;

service TestService {
    rpc simpleCall(TestRequest) returns (TestResponse) {}
}

message TestRequest {
    string text = 1;
}

message TestResponse {
    string text = 1;
}
```

Test case:
```java
package test;

import io.grpc.ManagedChannel;
import io.grpc.ManagedChannelBuilder;
import io.grpc.Server;
import io.grpc.netty.NettyServerBuilder;
import io.grpc.stub.StreamObserver;
import java.net.InetSocketAddress;

public class Test {

    static class TestService extends TestServiceGrpc.TestServiceImplBase {
        @Override
        public void simpleCall(TestRequest request, StreamObserver<TestResponse> responseObserver) {
            System.out.println("simple call received");
            String msg = request.getText();
            responseObserver.onNext(TestResponse.newBuilder().setText("Hello " + msg).build());
            responseObserver.onCompleted();
        }
    }

    public static void main(String[] args) throws Exception {
        InetSocketAddress socketAddr = new InetSocketAddress("localhost", 8080);
        NettyServerBuilder serverBuilder = NettyServerBuilder.forAddress(socketAddr);
        Server server = serverBuilder.addService(new TestService()).build();
        ManagedChannel channel = ManagedChannelBuilder.forAddress("localhost", 8080).usePlaintext().build();
        TestServiceGrpc.TestServiceBlockingStub stub = TestServiceGrpc.newBlockingStub(channel);

        try {
            TestResponse response = stub.simpleCall(TestRequest.newBuilder().setText("world").build());
            System.out.println(response.getText());
        } catch (Exception e) {
            // this is an expected exception
            System.out.println("exception received: " + e.getMessage());
        }
        
        server.start();
        
        // <-- here we get an exception that should not be raised
        TestResponse response = stub.simpleCall(TestRequest.newBuilder().setText("world 3").build());
        System.out.println(response.getText());
        
        server.shutdown();
    }

}
```
