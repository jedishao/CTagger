Planning bug  about MATCH_RECOGNIZE clause
The former issue raised by others is in slack: https://trinodb.slack.com/archives/C0305TQ05KL/p1667214424476489

Now we can reproduce this error 100% in Trino 392, 394 and 402 again, here is the method:  

```
USE hive.ta;

CREATE TABLE hive.ta.yyc_event_99 (                                                       
    "#user_id" bigint,                                                                                                                                 
    "#event_time" timestamp(3),                                                                                                                       
    "$part_event" varchar,                                                                
    "$part_date" varchar                                                                  
 )                                                                                        
 WITH (                                                                                   
    format = 'ORC',                                                                       
    partitioned_by = ARRAY['$part_event','$part_date']                                    
 );


insert into yyc_event_99 ("#user_id", "#event_time", "$part_event", "$part_date") 
values (1, timestamp '2022-11-29', 'vip_levelup', '2022-11-29');


SELECT virtual_user_id
FROM
  (
   SELECT
   "$part_event",
   "#user_id" virtual_user_id
   , "#event_time"
   FROM
     yyc_event_99
   WHERE "$part_event" = 'vip_levelup' AND "$part_date" = '2022-11-29')
  MATCH_RECOGNIZE (
   PARTITION BY virtual_user_id
   ORDER BY "#event_time" ASC
   ONE ROW PER MATCH
   AFTER MATCH SKIP PAST LAST ROW
   PATTERN (((A+) (B+)))
   DEFINE
      A AS ("$part_event" IN ('vip_levelup')),
      B AS (("$part_event" IN ('vip_levelup')) AND ("#event_time" > last(A."#event_time")))
);
```
