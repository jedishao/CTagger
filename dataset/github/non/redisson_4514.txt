@Service
public class RedisSubscribeService
{

   RedissonClient redissonClient;

   @Autowired
   public void setRedissonClient(RedissonClient redissonClient)
   {
      this.redissonClient = redissonClient;
   }

   @Bean
   public void subscribe()
   {
      RPatternTopic topic = redissonClient.getPatternTopic("__keyevent@0__:*", StringCodec.INSTANCE);
      topic.addListener(String.class, new PersistRedisListener());
   }

}

Configuration class : 



@Configuration
@AutoConfigureBefore(RedisAutoConfiguration.class)
@ConditionalOnClass({Redisson.class})
@EnableConfigurationProperties({ RedisProperties.class})
public class RedisConfiguration
{

   private final  RedisConfigData config = RedisConfigData.getInstance();

   /**
    * Instantiate Redis connection pool
    */

   @Bean
   @ConditionalOnMissingBean(RedissonClient.class)
   public RedissonClient redisson()
   {
      final String hostName = config.getProperty(RedisConfigKeys.REDIS_HOST_NAME);
      final int port = config.getInteger(RedisConfigKeys.REDIS_PORT_NUMBER);
      final String userName = config.getProperty(RedisConfigKeys.REDIS_USER_NAME);
      final Config redissonConfig = new Config();
      if (userName != null && !userName.isEmpty())
      {
         redissonConfig.useSingleServer().setAddress("redis://" + hostName + ":" + port).setUsername(userName)
                 .setPassword(config.getProperty(NamespaceRedisConfigKeys.REDIS_USER_PASSWORD));
      }
      else
      {
         redissonConfig.useSingleServer().setAddress("redis://" + hostName + ":" + port);
      }
      //redissonConfig.setCodec(StringCodec.INSTANCE);
      return Redisson.create(redissonConfig);
   }

}
