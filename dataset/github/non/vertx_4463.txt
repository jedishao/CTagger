JsonObject advanced function put()
~~~java
    /**
     * Put an Object into the JSON object with the specified key.
     * and require the value not null
     *
     * @param key   the key
     * @param value the value
     * @return a reference to this, so the API can be used fluently
     */
    public JsonObject putNotNull(String key, Object value) {
        Objects.requireNonNull(key);
        if (value != null) {
            map.put(key, value);
        }
        return this;
    }

    /**
     * Put an Object into the JSON object with the specified key.
     * to fix the json with key value
     *
     * @param key   the key
     * @param value the value
     * @return a reference to this, so the API can be used fluently
     */
    public JsonObject fix(String key, Object value) {
        Objects.requireNonNull(key);
        map.putIfAbsent(key, value);
        return this;
    }
~~~
