How to join based on the date and hour values of __time
This is my Druid sql query 

```
SELECT TIMESTAMPADD(HOUR,time_extract(t1.__time,'Hour') ,DATE_TRUNC('DAY',t1.__time))  AS __timestamp,
          (max_val1 - min_val1) AS VAL1,
          (max_val2 - min_val2) AS VAL2,
          (max_val3 - min_val3) AS VAL3,
          t1.UID AS UID
   FROM
     (SELECT __time,
             VAL1 AS max_val1,
             UID,
             VAL2 AS max_val2,
             VAL3 AS max_val3
      FROM one_month
      WHERE __time > TIME_PARSE('2020-09-26','yyyy-MM-dd')
        and __time in
          (SELECT MAX(__time)
           FROM one_month
           WHERE __time > TIME_PARSE('2020-09-26','yyyy-MM-dd')
           GROUP BY TIMESTAMPADD(HOUR,time_extract(__time,'Hour') ,DATE_TRUNC('DAY',__time)) ,
                    UID) ) AS t1
   INNER JOIN
     (SELECT __time ,
             VAL1 AS min_val1,
             UID,
             VAL2 AS min_val2,
             VAL3 AS min_val3
      FROM one_month
      WHERE __time > TIME_PARSE('2020-09-26','yyyy-MM-dd')
        and __time in
          (SELECT MIN(__time)
           FROM one_month
           WHERE __time > TIME_PARSE('2020-09-26','yyyy-MM-dd')
           GROUP BY TIMESTAMPADD(HOUR,time_extract(__time,'Hour') ,DATE_TRUNC('DAY',__time)) ,
                    UID) ) AS t2 ON DATE_TRUNC('HOUR',t2.__time) =DATE_TRUNC('HOUR',t1.__time) AND t1.UID = t2.UID
```
