Duplicate subscriptions with RedisMessageListenerContainer in Spring Data 2.7
There is a serious problem with Redisson and Spring Data 2.7. Each listener registered using `RedisMessageListenerContainer` produces 2 subscriptions.

**Expected behavior**

Each Listener for a channel or pattern is mapped to a single subscription.

**Actual behavior**

Listeners for channels and patterns produce duplicate subscriptions.

**Steps to reproduce or test case**

The following test-case passes with Spring Data 2.6, but fails with 2.7 because there are duplicate listeners:

```java
import org.awaitility.Awaitility;
import org.junit.jupiter.api.Test;
import org.redisson.Redisson;
import org.redisson.api.RedissonClient;
import org.redisson.client.ChannelName;
import org.redisson.client.RedisPubSubListener;
import org.redisson.pubsub.PubSubConnectionEntry;
import org.redisson.pubsub.PublishSubscribeService;
import org.redisson.spring.data.connection.RedissonConnectionFactory;
import org.springframework.data.redis.connection.MessageListener;
import org.springframework.data.redis.listener.ChannelTopic;
import org.springframework.data.redis.listener.PatternTopic;
import org.springframework.data.redis.listener.RedisMessageListenerContainer;
import org.springframework.test.context.ActiveProfiles;
import org.springframework.test.context.junit.jupiter.SpringJUnitConfig;

import javax.inject.Inject;

import java.time.Duration;
import java.util.Arrays;
import java.util.Collections;
import java.util.Queue;

import static org.assertj.core.api.Assertions.assertThat;
import static org.awaitility.Awaitility.await;

class RedissonListenerContainerTest {

	@Inject
	private RedissonClient client;

	@Test
	void shouldRegisterListeners() {
		final MessageListener aListener = (message, pattern) -> {

		};
		final RedisMessageListenerContainer c = newContainer(client, aListener);
		c.afterPropertiesSet();
		c.start();
		
		final PublishSubscribeService subscribeService = ((Redisson) client).getConnectionManager().getSubscribeService();
		final String channelName = "a";
		Awaitility.await()
				.atMost(Duration.ofSeconds(5))
				.until(() -> subscribeService.getPubSubEntry(new ChannelName(channelName)) != null);
		final PubSubConnectionEntry entry = subscribeService.getPubSubEntry(new ChannelName(channelName));
		final Queue<RedisPubSubListener<?>> listeners = entry.getListeners(new ChannelName(channelName));
		assertThat(listeners).hasSize(1);
	}

	public RedisMessageListenerContainer newContainer(RedissonClient client, MessageListener messageListener) {
		final RedisMessageListenerContainer container = new RedisMessageListenerContainer();
		container.setConnectionFactory(new RedissonConnectionFactory(client));
		container.addMessageListener(messageListener, 
				Arrays.asList(new ChannelTopic("a"), new ChannelTopic("b")));
		container.addMessageListener(messageListener,
				Arrays.asList(new PatternTopic("c*")));
		return container;
	}
}
```

**Redis version**

7.0.2

**Redisson version**

3.17.4

**Redisson configuration**

```java
final Config cfg = new Config();
cfg.useSingleServer()
        .setAddress(redisAddress)
        .setPassword(Strings.emptyToNull(redisPassword))
        .setTimeout(redisTimeout)
        .setConnectTimeout(redisConnectTimeout)
        .setRetryAttempts(redisRetryAttempts)
        .setRetryInterval(redisRetryInterval)
        .setPingConnectionInterval(pingConnectionInterval)
        .setConnectionPoolSize(redisConnectionPoolSize)
        .setConnectionMinimumIdleSize(redisConnectionPoolMinIdleSize)
        .setSubscriptionConnectionPoolSize(redisSubscriptionConnectionPoolSize)
        .setSubscriptionConnectionMinimumIdleSize(redisSubscriptionConnectionPoolMinIdleSize);
return Redisson.create(cfg);
```
