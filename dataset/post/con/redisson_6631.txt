``` java
package cn.ling.redislock.job;

import lombok.extern.slf4j.Slf4j;
import org.redisson.api.RLock;
import org.redisson.api.RedissonClient;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.context.annotation.Lazy;
import org.springframework.stereotype.Service;

import javax.annotation.PostConstruct;
import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicInteger;

@Service
@Slf4j
public class SingleThreadPoolDemo {

    private @Autowired @Lazy RedissonClient redissonClient;
    private static ExecutorService executorService;
    @PostConstruct
    public void begin() {
        AtomicInteger num = new AtomicInteger(0);
        executorService = new ThreadPoolExecutor(
                2, 2, 0L, TimeUnit.MILLISECONDS, new LinkedBlockingQueue<>(),
                r -> {
                    Thread thread = new Thread(r, "loop_thread_" + num.incrementAndGet());
                    thread.setDaemon(false);
                    return thread;
                }
        );
        for (int i = 0; i < 2; i++) {
            startOneTask();
        }
    }

    private void startOneTask() {
        executorService.submit(() -> {
            RLock lock = null;
            while (true) {
                try {
                    TimeUnit.SECONDS.sleep(1);
                    lock = redissonClient.getLock("testLock-ling");
                    log.info("to get lock");
                    lock.tryLock(20, 30, TimeUnit.SECONDS);
                    log.info("get lock");
                } catch (Exception e) {
                    log.info("try to get lock ex:", e);
                } finally {
                    try {
                        assert lock != null;
                        if (lock.isLocked() && lock.isHeldByCurrentThread()) {
                            log.info("to unlock");
                            lock.unlock();
                            log.info("unlock ok");
                        }
                    } catch (Exception e) {
                        log.info("unlock ex", e);
                    }
                }
            }
        });
    }
}
```

Finally, the threads are both WAITING(PARKING) state !
