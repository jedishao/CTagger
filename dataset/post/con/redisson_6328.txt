```
protected void processTask() {
    while (true){
        RLock resenfKafkaLock = null;
        boolean b = false;
        try {
            // Acquire lock from Redis
            RedissonClient redisClient = dynamicRedis.getRedisClient();
            resenfKafkaLock = redisClient.getLock(LockConstants.LOCK_RESEND);
            b = resenfKafkaLock.tryLock(10, 600, TimeUnit.SECONDS);
            if (b) {
                // Process tasks...
                // Return if no tasks
            }
        } catch (Exception ex) {
            logger.error("resend error {}", ex.toString());
        } finally {
            LockUtils.release(resenfKafkaLock, b);
        }
    }
}
```
