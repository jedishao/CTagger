            new ReturnAttribute("score"),
            new ReturnAttribute("model")
    );
    public List<Result> search(byte[] data, int topK) {
        RSearch s = client.getSearch(
        rnew CompositeCodec(new StringCodec(), new StringCodec());
);

        topK = (topK > 0) ? topK : this.topK;
        List<Result> results = new ArrayList<>();
        try {

            String command =
                    "*=>[KNN " + this.dimension + " @vector $BLOB AS score ]";
            Map<String, Object> params = new HashMap<>(2);

            params.put("TOPK", topK);
            params.put("BLOB", data);

            SearchResult search = s.search(this.collectionName, command, QueryOptions.defaults()
                    .sortBy("score")
                    .limit(0, topK > 0 ? topK : this.topK)
                    .dialect(2)
                    .params(params)
                    .returnAttributes(returnAttributes)
            );

            if (search.getTotal() > 0) {
                for (org.redisson.api.search.query.Document result : search.getDocuments()) {
                    if (result != null) {
                        results.add(new Result(result.getId(), (Double) result.getAttributes().get("score")));
                    }
                }
            } else {
                log.warn("search not result");
            }
        } catch (Exception e) {
            log.error("Search failed: " + e);
            throw e;
        }
        return results;
    }
