Optimize Iceberg partitions doesn't work when filtering by "$partition" on multiple partitioned columns phrase
```sql
-----------------------------------
-- Single partitioned column works 
-----------------------------------

CREATE TABLE test_table (
    c1 integer,
    c2 date,
    c3 double)
WITH (
    format = 'PARQUET',
    partitioning = ARRAY['month(c2)']);

INSERT INTO test_table values (1,date'2025-07-13', 0.1), (3,date'2025-08-01', 0);
INSERT INTO test_table values (2,date'2025-07-13', 0.1), (4,date'2025-08-01', 0);

SELECT partition,
       reduce(regexp_extract_all(file_path,'([a-z0-9_]+)=[^/]+[/]'), map(), (s, e) -> map_concat(s, map([split(e,'=')[1]], [rtrim(split(e,'=')[2],'/')])), s->s) file_partitions,
       count(*) files
FROM "test_table$files" GROUP BY AUTO;

|partition     |file_partitions   |files|

|{c2_month=666}|{c2_month=2025-07}|2    |
```
