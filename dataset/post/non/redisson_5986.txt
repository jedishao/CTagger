### Redisson configuration

public RedissonClient init(RedissionProperties properties) {
		
		String[] sentinelNodes = convertSentinalIntoArray(
				Arrays.asList(((String) properties.getSentinelNodes()).split(Symbol.COMMA_STRING)));
		
		LOG.info("Decypted Sentinel Servers :First Sentinel , Second Sentinal "+ sentinelNodes[0]
				+ sentinelNodes[1]);

		Config config = new Config();
		config.useSentinelServers().setMasterName(properties.getMaster())
				.setDatabase(Integer.parseInt(properties.getIndex())).addSentinelAddress(sentinelNodes)
				.setPassword(properties.getChecksum()).setConnectTimeout(30000) 
				.setTimeout(10000) 
				.setRetryAttempts(3) 
				.setRetryInterval(1000) 
				.setMasterConnectionPoolSize(20) 
				.setSlaveConnectionPoolSize(20) 
				.setMasterConnectionMinimumIdleSize(10) 
				.setSlaveConnectionMinimumIdleSize(10)
				.setKeepAlive(true);
		
		config.setThreads(2);
		config.setNettyThreads(2);
		
		Codec codec=new JsonJacksonCodec();	
		config.setCodec(codec);
		
		config.setLockWatchdogTimeout(DEFAULT_LOCK_WATCHDOG_TIMEOUT);
		LOG.info("Redission Client Bean initialized successfully.");
		return Redisson.create(config);
	}
