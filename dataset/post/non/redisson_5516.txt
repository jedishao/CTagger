@Service
@Slf4j
public class TestIntegration {

    @Resource
    private TestPassApi testPassApi;

    @Resource
    private RedissonClient redissonClient;

    /**
     * limiter
     */
    RRateLimiter testLimiter;

    @PostConstruct
    public void init() {
        testLimiter = redissonClient.getRateLimiter("test-limiter");
        testLimiter.trySetRate(RateType.OVERALL, 1, 1, RateIntervalUnit.SECONDS);
    }

    public Boolean isPass(String testContent) {
        JSONObject jsonObject = new JSONObject();
        jsonObject.put("test_content", testContent);
        RequestBody dto = RequestBody.create(MediaType.get("application/json"), jsonObject.toJSONString());
        Retryer<Boolean> retryer = RetryUtils.getFixedWaitRetryer(1, TimeUnit.SECONDS, 3);
        try {
            return retryer.call(() -> {
                log.info("isPass req: {}", jsonObject.toJSONString());
                // limit 1s 1request
                if (!testLimiter.tryAcquire(1, TimeUnit.MINUTES)) {
                    log.warn("testLimiter limiter timeout");
                    return false;
                }
                var call = testPassApi.test(dto).execute();
                return true;
            });
        } catch (Exception e) {
            log.error("occur error. req: {}", jsonObject.toJSONString(), e);
            return false;
        }
    }
}
