Trino Server crashes when UDF makes socket connection via JNI
I am testing on Starburst Trino 364-e.2 Single Node cluster. I had previously tested the same with Starburst Trino 356-e (where the Trino server was crashing too). 

OS: Amazon Linux 2

JRE Version:
```
openjdk 11.0.12 2021-07-20 LTS
OpenJDK Runtime Environment Corretto-11.0.12.7.1 (build 11.0.12+7-LTS)
OpenJDK 64-Bit Server VM Corretto-11.0.12.7.1 (build 11.0.12+7-LTS, mixed mode)
```

I have a custom UDF that makes a makes a call to a native method via JNI layer. The native library tries to establish a TCP connection to a fixed non-blocking socket localhost:15780.  When this happens, the Trino server aborts without any error log in server.log. But the launcher.log prints
`*** stack smashing detected ***: <unknown> terminated`

Here's the Custom UDF Java Source Code (com/example/test/ptyTestUDF.java):
```
package com.example.test;

import io.airlift.log.Logger;
import io.airlift.slice.Slice;
import io.airlift.slice.Slices;
import io.trino.spi.connector.ConnectorSession;
import io.trino.spi.function.Description;
import io.trino.spi.function.ScalarFunction;
import io.trino.spi.function.SqlNullable;
import io.trino.spi.function.SqlType;
import io.trino.spi.type.StandardTypes;

public class ptyTestUDF {
    private static Logger logger = Logger.get(ptyTestUDF.class);
    @SqlNullable
    @ScalarFunction( "ptyTestUDF" )
    @Description( "Establishes a socket connection to localhost:15780 via JNI bridge and writes a JSON to it" )
    @SqlType( StandardTypes.VARCHAR )
    public static Slice ptyTest( ConnectorSession session ) {
        HelloWorldJNI helloWorldJNI = new HelloWorldJNI();

        logger.info("Calling native sayHello()");
        helloWorldJNI.sayHello();
        logger.info("Called native sayHello()");

        return Slices.utf8Slice( "Done" );
    }
}

Here's the C++ implementation of the native method that makes socket connection using the send() function (com_example_test_HelloWorldJNI.cpp):
```
#include "com_example_test_HelloWorldJNI.h"
#include <iostream>
#include <fstream>
#include <fcntl.h>
#include <sys/socket.h>
#include <errno.h>
#include <sys/un.h>
#include <netdb.h>
#include <arpa/inet.h>
#include <unistd.h>
#include <string.h>
#define PORT 15780

using namespace std;



void send( char* msg, int dataLen, char* host, int port )
{
  int sd                    = -1;
  int errorCode             = 0;
  int saveFlags             = 0;
  int len                   = -1;
  int sendLen               = 0;
  int numSent               = 0;
  unsigned int nonBlocking  = 1;
  socklen_t     nLength     = 0;
  struct sockaddr_in sa;
  struct hostent* pHostEntry;
  struct timeval stTimeout;
  fd_set fdConnect, fdError;


  /* P_createSocketEx */
  sd = socket( 2, SOCK_STREAM, 0 );
  if( -1 == sd )
  {
    printf( "P_createSocketEx failed: sd=%d, errno=%d\n", sd, errno );
    return;
  }
  
  /* P_setSocketNonBlocking */
  saveFlags = fcntl( sd, F_GETFL, 0 );
  if( saveFlags < 0 )
  {
    printf( "P_setSocketNonBlocking fcntl 1 failed: saveFlags=%d, errno=%d\n", saveFlags, errno );
    return;
  }
  saveFlags |= O_NONBLOCK;
  if( fcntl( sd, F_SETFL, saveFlags ) < 0 )
  {
    printf( "P_setSocketNonBlocking fcntl 2 failed: saveFlags=%d, errno=%d\n", saveFlags, errno );
    return;
  }

  /* P_connect */
  memset( &sa, 0, sizeof( sa ) );
  sa.sin_family = AF_INET;
  sa.sin_port   = htons( ( unsigned short )port );

  /* first try with a.b.c.d format */
  sa.sin_addr.s_addr = inet_addr( host );
  if( INADDR_NONE == sa.sin_addr.s_addr )
  {
    /* try with host name */
    pHostEntry = gethostbyname( host );
    if( NULL == pHostEntry )
    {
      printf( "gethostbyname failed: %d\n", saveFlags );
      return;
    }
    memcpy( &sa.sin_addr.s_addr,
              pHostEntry->h_addr_list[0],
              pHostEntry->h_length );
  }
  saveFlags = fcntl( sd, F_GETFL, 0 );
  if( saveFlags < 0 )
  {
    printf( "gethostbyname failed: saveFlags=%d, errno=%d\n", saveFlags, errno );
    return;
  }

  /* Set non blocking */
  if( fcntl( sd, F_SETFL, saveFlags|O_NONBLOCK ) < 0 )
  {
    printf( "fcntl failed: %d\n", errno );
    return;
  }
  errno = 0;

  /* Try the connect */
  if( ( connect( sd, ( struct sockaddr* ) &sa, sizeof( sa ) ) ) == -1 )
  {    
    /*printf("connect failed: %d\n", errno);*/
  }
  
  /* Since socket is nonblocking check if in progress */
  if( EINPROGRESS == errno )
  {
    FD_ZERO( &fdConnect );
    FD_ZERO( &fdError);
    FD_SET( sd, &fdConnect );
    FD_SET( sd, &fdError);
    stTimeout.tv_sec   = 10;
    stTimeout.tv_usec  = 0;

    errorCode = select( ( int )sd + 1, NULL, &fdConnect, &fdError, &stTimeout );

    /* Get the return code from the connect */
    /* errorcode=0 means success, otherwise it contains the errno. */
    nLength = sizeof( errorCode );
    if( getsockopt( sd, SOL_SOCKET, SO_ERROR, ( char* )&errorCode, &nLength ) != 0 && 0 != errorCode )
    {
      printf( "getsockopt failed: errno=%d\n", errno );
      return;
    }
  }
  /* Restore flags */
  if( fcntl( sd, F_SETFL, saveFlags ) < 0 )
  {
    printf( "fcntl failed: saveFlags=%d, errno=%d\n", saveFlags, errno );
    return;
  }

  /* P_send */
  do
  {
    sendLen = ( int )dataLen - numSent;

    len = send( sd, ( char* )( msg + numSent ), sendLen, 0 );

    if( len == -1 )
    {
      printf( "send failed: len=%d, errno=%d\n", len, errno );
      return;
    }

    if( len == 0 )
    {
      printf( "disconnected: %d\n", errno );
      return;
    }

    numSent += len;
  }
  while( len > 0 && numSent < ( int )dataLen );

  /* P_closeSocket */
  shutdown( sd, SHUT_RDWR );
  if( -1 == close( sd ) )
  {
    printf( "close failed: %d\n", errno );
    return;
  }
}

/******************************************************************************************************/

  (JNIEnv *, jobject)

  cout << "Hello from C++ !!" << endl;
  char msg[]  = "{\"key\": \"This is from ptyTestUDF\"}";
  char host[] = "localhost";
  int port    = 15780;

  send( msg, sizeof(msg), host, port );

How to trigger the issue: 

Start a TCP listener using ncat on a terminal:
```
$ nc -4 -k -l 15780
```

Run Test UDF via Trino CLI
```
trino> select ptyTestUDF();
```
After this the Trino Server crashes.
