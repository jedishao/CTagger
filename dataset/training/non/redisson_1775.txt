        }

    }

    public static class ThreadDemo extends Thread {

        private Thread t;
        private String threadName;

        ThreadDemo( String name) {
            threadName = name;
        }

        public void run() {
            try {
                log.info(Thread.currentThread().getName() + " start  "); 
                
                Config config = new Config(); 
                config.setCodec(new org.redisson.client.codec.StringCodec()); 
                config.useSingleServer().setAddress("redis://10.24.51.167:6380"); 
                RedissonClient redisson = Redisson.create(config);

                RLock lock = redisson.getFairLock("lock");
                
                boolean res =  lock.tryLock(1, 10, TimeUnit.SECONDS);
                if(res){
                    log.info("thread {} get lock success, sleep 5s",Thread.currentThread().getName());
                    Thread.sleep(5000); 
                    if(lock.isHeldByCurrentThread()){
                        lock.unlock();
                    }
                    log.info("thread {} unlock  ",Thread.currentThread().getName());
                }else{
                    log.info("thread {} get lock faild",Thread.currentThread().getName());
                }
                log.info(Thread.currentThread().getName() + " end ");
            }catch (Exception e){
                e.printStackTrace();
                log.info(e.getMessage());
            }
        }

        public void start () {
            if (t == null) {
                t = new Thread (this, threadName);
                t.start ();
            }
        }
    }
}`
