example code:
```
import org.apache.commons.io.IOUtils;
import org.redisson.Redisson;
import org.redisson.api.RedissonClient;
import org.redisson.api.listener.MessageListener;
import org.redisson.api.listener.PatternMessageListener;
import org.redisson.client.codec.StringCodec;
import org.redisson.config.Config;

import java.io.InputStream;
import java.nio.charset.StandardCharsets;

public class redissonTest {
	private static RedissonClient redisson;
	public static void main(String[] args) throws Exception {
		initRedisson("redis.yaml");

		initExpiredTopic();
		System.out.println("wait sub...");
		while (true) {
			Thread.sleep(5000);
		}

	}

	public static void initRedisson(String file) throws Exception {
		try (InputStream in = redissonTest.class.getResourceAsStream(file)) {
			String yamlConfig = IOUtils.toString(in, StandardCharsets.UTF_8);
			Config config = Config.fromYAML(yamlConfig);

			redisson = Redisson.create(config);
		}
	}

	public static class ExpiredListener implements MessageListener<String> {
		@Override
		public void onMessage(CharSequence channel, String msg) {
			System.out.printf("onExpired::key:%s, charSequence:%s\n", msg, channel);
		}
	}

	public static class PatternExpiredListener implements PatternMessageListener<String> {
		@Override
		public void onMessage(CharSequence pattern, CharSequence channel, String msg) {
			System.out.printf("onExpired::key:%s, pattern:%s, charSequence:%s\n", msg, pattern, channel);
		}
	}

	private static void initExpiredTopic() {
		System.out.println("register topic");
		redisson.getTopic("__keyevent@0__:expired", StringCodec.INSTANCE)
				.addListener(String.class, new ExpiredListener());
		System.out.println("register pattern topic");
		redisson.getPatternTopic("__keyevent*:expired", StringCodec.INSTANCE)
				.addListener(String.class, new PatternExpiredListener());
	}
}
```
